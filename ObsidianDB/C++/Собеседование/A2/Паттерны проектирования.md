## Что такое паттерн Singleton? Какие проблемы он создаёт, в том числе в многопоточном коде?

Пример реализации Singleton стандартной версии и Скотта Мейерса:
```C++
class ExampleSingleton {
public:
  ExampleSingleton(const ExampleSingleton &) = delete;
  ExampleSingleton &operator=(const ExampleSingleton &) = delete;
  ExampleSingleton(ExampleSingleton &&) = delete;
  ExampleSingleton &operator=(ExampleSingleton &&) = delete;

  static ExampleSingleton &getInstance(std::string name) {
    std::lock_guard<std::mutex> lock(mtx_);

    if (!instance_) {
      instance_ = std::unique_ptr<ExampleSingleton>(
          new ExampleSingleton(std::move(name)));
    }

    return *instance_;
  }

  void doSome() const {
    std::cout << std::this_thread::get_id() << " - " << name_ << '\n';
  }

private:
  explicit ExampleSingleton(std::string name) : name_(std::move(name)) {}

  ~ExampleSingleton() = default;

  static std::unique_ptr<ExampleSingleton> instance_;
  static std::mutex mtx_;

  std::string name_;
};

// Обязательная инициализация
ExampleSingleton *ExampleSingleton::instance_ = nullptr;
std::mutex ExampleSingleton::mtx_;

class MeyersSingleton {
  /**
   * @brief Пример синглтона Скотта Мейерса
   *  Объект создается на месте, что начиная с C++11 потокобезопасно
   */
public:
  MeyersSingleton(const MeyersSingleton &) = delete;
  MeyersSingleton &operator=(const MeyersSingleton &) = delete;
  MeyersSingleton(MeyersSingleton &&) = delete;
  MeyersSingleton &operator=(MeyersSingleton &&) = delete;

  static MeyersSingleton &getInstance(std::string name) {
    static MeyersSingleton instance_(std::move(name));
    return instance_;
  }

  void doSome() {
    std::cout << std::this_thread::get_id() << " - " << name_ << '\n';
  }

private:
  explicit MeyersSingleton(std::string name) : name_(std::move(name)) {}
  ~MeyersSingleton() = default;

  std::string name_;
};

void ThreadBar(std::string name_) {
  // ExampleSingleton / MeyersSingleton
  MeyersSingleton::getInstance(name_).doSome();
}

int main() {

  auto t1 = std::thread(ThreadBar, "t1");
  auto t2 = std::thread(ThreadBar, "t2");

  t1.join();
  t2.join();

  return 0;
}
```
