```C++
class Solution {
public:
  bool isPalindrome(ListNode *head) {
    if (head == nullptr) {
      return true;
    }

    std::stack<ListNode *> nodesStack;
    auto currentPtr = head;
    while (currentPtr != nullptr) {
      nodesStack.push(currentPtr);
      currentPtr = currentPtr->next;
    }

    auto startPtr = head;
    auto endPtr = nodesStack.top();

    auto start = 0;
    auto end = nodesStack.size() - 1;

    while (start < end) {
      if (startPtr->val != endPtr->val) {
        return false;
      }
      startPtr = startPtr->next;
      ++start;
      nodesStack.pop();
      endPtr = nodesStack.top();
      --end;
    }
    return true;
  }
};
```

```js
function isPalindrome(head: ListNode | null): boolean {
    if (head === null) {
        return false;
    }
    let currentNode: ListNode | null = head;
    let nodes: ListNode[] = [];
    while (currentNode != null) {
        nodes.push(currentNode);
        currentNode = currentNode.next;
    }

    let start = 0;
    let end = nodes.length - 1;

    while (start < end) {
        if (nodes[start].val != nodes[end].val) {
            return false;
        }
        ++start;
        --end;
    }
    return true;
};
```
